Integration Guide

Add the assistant to your website or application in seconds.

Option 1: Script Tag (HTML / CMS / WordPress / Webflow)

<!-- Add to <head> or before </body> -->
<script
  src="https://your-assistant-domain.com/sdk/assistant.js"
  data-assistant-id="default"
  data-theme="auto"
  data-position="bottom-right"
  defer
></script>

Option 2: Programmatic JavaScript API

// Initialize programmatically
await AssistantSDK.init({
  assistantId: 'researcher',
  theme: 'dark',
  position: 'bottom-right',
  primaryColor: '#0284c7'
});

// Listen to conversation events
AssistantSDK.on('message:received', (msg) => {
  console.log('Assistant replied:', msg.content);
});

// Control widget
AssistantSDK.open();
AssistantSDK.sendMessage('Please summarize this document.');

Option 3: React / Next.js Component

import { useEffect } from 'react';

export function AIAssistantWidget() {
  useEffect(() => {
    const script = document.createElement('script');
    script.src = 'https://your-assistant-domain.com/sdk/assistant.js';
    script.setAttribute('data-assistant-id', 'default');
    script.async = true;
    document.body.appendChild(script);

    return () => {
      if (window.AssistantSDK) {
        window.AssistantSDK.destroy();
      }
      document.body.removeChild(script);
    };
  }, []);

  return null;
}
